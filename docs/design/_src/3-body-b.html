04 · 色彩系统

颜色从种子色和封面里长出来

用 com.materialkolor:material-kolor 生成整套 Material 颜色角色。全局主题来自用户选的种子色（或 Android 12+ 的壁纸色），详情页、播放器和刷刷再用当前视频封面的颜色局部覆盖一层。这一页上的所有颜色都由同一套算法（material-color-utilities，material-kolor 移植的就是它）在浏览器里实时算出来，改种子色或风格就能看到效果。

// 全局主题：用户种子色 / 壁纸色，PaletteStyle 和对比度来自设置
@OptIn(ExperimentalMaterial3ExpressiveApi::class)
@Composable fun BvTheme(prefs: ThemePrefs = LocalThemePrefs.current, content: @Composable () -> Unit) {
    val dark = prefs.darkMode.resolve(isSystemInDarkTheme())
    val seed = if (prefs.source == ThemeSource.Wallpaper) wallpaperSeedOr(prefs.seed) else prefs.seed
    DynamicMaterialExpressiveTheme(
        seedColor = seed,
        isDark = dark,
        isAmoled = prefs.pureBlack,                       // 影院纯黑
        style = prefs.paletteStyle,                       // 默认 PaletteStyle.Vibrant：保留品牌色相、饱和度高
        contrastLevel = prefs.contrast,
        specVersion = ColorSpec.SpecVersion.SPEC_2025,
        motionScheme = MotionScheme.expressive(),
        typography = bvTypography(LocalDeviceProfile.current),
        animate = true,
        content = content,
    )
}

// 局部主题：详情 / 播放器 / 刷刷。封面颜色在后台线程算，按 URL 缓存
@Composable fun CoverTheme(cover: String?, content: @Composable () -> Unit) {
    val seed by rememberCoverSeed(cover)              // null 时沿用全局 primary
    DynamicMaterialExpressiveTheme(
        seedColor = seed ?: MaterialTheme.colorScheme.primary,
        isDark = true, style = PaletteStyle.Fidelity, animate = true, content = content,
    )
}

suspend fun CoverSeedCache.get(url: String): Color? = cache.getOrPut(url) {
    val bitmap = imageLoader.execute(
        ImageRequest.Builder(context).data(url).size(112).allowHardware(false).build()
    ).drawable?.toBitmap()?.asImageBitmap() ?: return null
    withContext(Dispatchers.Default) { bitmap.themeColors(fallback = Color.Unspecified).firstOrNull() }
}
05 · 组件与动效

Expressive 用在状态变化上，不用来装饰

形状变形、弹簧动画和强调字重只在按下、聚焦、选中、加载这些状态变化时出现。静止画面保持干净，让封面和视频做主角。

加载与焦点

LoadingIndicator：七种形状边转边变。用于首屏加载、下拉刷新（PullToRefreshBox + LoadingIndicator）、播放器缓冲，替换现在的线性进度条。

焦点：放大 1.08、3dp primary 描边 + 同色光晕、圆角 16→22dp。动画曲线由页面脚本按 MotionScheme.expressive() 的弹簧参数实时生成。把鼠标移到卡片上，或按 Tab 键试试。
