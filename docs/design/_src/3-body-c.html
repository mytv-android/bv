07 · 新入口「刷刷」

一直往下刷，刷到的每条都带着自己的颜色

// designsystem/focus/BvFocus.kt（示意；正式实现用 Modifier.Node）
fun Modifier.bvFocus(
    shape: CornerBasedShape = RoundedCornerShape(16.dp),
    focusedShape: CornerBasedShape = RoundedCornerShape(22.dp),
    scale: Float = 1.08f,
) = composed {
    val profile = LocalDeviceProfile.current
    var focused by remember { mutableStateOf(false) }
    // 触摸模式下不放大不描边，只有方向键 / 键盘导航时才显示
    val visible = focused && profile.input != InputMode.Touch
    val motion = MaterialTheme.motionScheme
    val s by animateFloatAsState(if (visible) scale else 1f, motion.fastSpatialSpec())
    val morph by animateFloatAsState(if (visible) 1f else 0f, motion.defaultSpatialSpec())
    val ring = MaterialTheme.colorScheme.primary
    this
        .onFocusChanged { focused = it.isFocused }
        .graphicsLayer { scaleX = s; scaleY = s }
        .focusRing(visible, ring, lerp(shape, focusedShape, morph))
        .focusable()
}
